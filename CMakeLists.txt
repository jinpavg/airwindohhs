cmake_minimum_required(VERSION 3.28)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

project(airwindohhs VERSION 0.1.0)

add_library(Airwindohhs INTERFACE)
add_library(Airwindohhs::Airwindohhs ALIAS Airwindohhs)
target_include_directories(Airwindohhs INTERFACE include)
target_compile_features(Airwindohhs INTERFACE cxx_std_17)

option(AIRWINDOHHS_BUILD_TOOLS "Build catalog tools" ON)
option(AIRWINDOHHS_BUILD_STATE_TESTS "Build deterministic state tests" OFF)
if(AIRWINDOHHS_BUILD_TOOLS)
    add_executable(dump_meta tools/dump_meta.cpp)
    target_link_libraries(dump_meta PRIVATE Airwindohhs::Airwindohhs)
endif()
if(AIRWINDOHHS_BUILD_STATE_TESTS)
    enable_testing()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(seeded_state_tests test/seeded_state.cpp)
    target_link_libraries(seeded_state_tests PRIVATE Airwindohhs::Airwindohhs)
    if(MSVC)
        target_compile_options(seeded_state_tests PRIVATE /W4 /WX /UNDEBUG)
    else()
        # Existing upstream pointer-advance expressions intentionally discard values.
        target_compile_options(seeded_state_tests PRIVATE -Wall -Wextra -Werror -Wno-unused-value -UNDEBUG)
    endif()
    add_test(NAME seeded_state COMMAND seeded_state_tests)
    add_test(NAME seeded_generation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_seeded_generation.py)
endif()
